% Oracle cases for lane r/grafik-3 (colour order, stem markers, subplot
% insets, colorbar next to a fixed-aspect box, colormaps by name).
% Measure: GRAFIK_TOOLKIT=qtnative GRAFIK_CASES=<this file> motor/scripts/grafik_oracle.sh
% (native Qt, as the IndyMAT app runs Octave: its text extents decide the
% subplot insets; fltk measures titles about 1 px taller).
% tests/oracle_g3.rs runs each body through the interpreter and compares.

%% co_spec_single
t=0:3; plot(t,t,'k',t,2*t,t,3*t)
%% co_spec_matrix
t=0:3; plot(t,[t' 2*t'],'r',t,t)
%% co_spec_repeat
t=0:3; plot(t,t,'k',t,2*t,'k',t,3*t)
%% co_hold_sequence
t=0:3; hold on; plot(t,t,'k'); plot(t,2*t); plot(t,3*t,'Color',[1 0 0]); plot(t,4*t); hold off
%% co_hold_spec_twice
t=0:3; hold on; plot(t,t,'k'); plot(t,t,'r'); plot(t,2*t); hold off
%% co_color_option_then_hold
t=0:3; plot(t,t,'Color',[1 0 0]); hold on; plot(t,2*t); hold off
%% co_antiwindup
t=0:3; plot(t,t,'k--',t,[t' 2*t']); legend('Hedef','a','b')
%% co_line_color
t=0:3; hold on; line(t,t,'Color','k'); plot(t,t); hold off
%% co_stem_stairs_spec
t=0:3; hold on; stem(t,t,'k'); stairs(t,t,'k'); plot(t,t); hold off
%% stem_color_option
t=0:3; stem(t,t,'Color',[0.9 0.4 0.15])
%% stem_spec_color
t=0:3; stem(t,t,'r')
%% stem_filled_color
t=0:3; stem(t,t,'filled','Color',[1 0 0])
%% stem_set_color
t=0:3; h=stem(t,t); set(h,'Color',[1 0 0])
%% sp_title_both
for p=1:2, subplot(2,1,p); plot(1:3); title('T'); end
%% sp_title_bottom
subplot(2,1,1); plot(1:3); subplot(2,1,2); plot(1:3); title('T')
%% sp_title_top
subplot(2,1,1); plot(1:3); title('T'); subplot(2,1,2); plot(1:3)
%% sp_xlabel_title_3x1
for p=1:3, subplot(3,1,p); plot(1:3); xlabel('X'); title('T'); end
%% sp_title_two_lines
for p=1:2, subplot(2,1,p); plot(1:3); title({'T','U'}); end
%% sp_title_fs16
for p=1:2, subplot(2,1,p); plot(1:3); xlabel('X','FontSize',16); title('T','FontSize',16); end
%% sp_2x2_ax4
for p=1:4, subplot(2,2,p); plot(1:3); end; title('T')
%% sp_xlabel_5lines
for p=1:2, subplot(2,1,p); plot(1:3); xlabel({'X','Y','Z','W','V'}); end
%% sp_4x1_xlabel_3lines
for p=1:4, subplot(4,1,p); plot(1:3); xlabel({'X','Y','Z'}); end
%% sp_ylabel_6lines
subplot(1,2,1); plot(1:3); subplot(1,2,2); plot(1:3); ylabel({'A','B','C','D','E','F'})
%% sp_tall_figure
set(gcf,'Position',[0 0 560 800]); for p=1:2, subplot(2,1,p); plot(1:3); title('T'); end
%% sp_short_figure
set(gcf,'Position',[0 0 560 300]); for p=1:2, subplot(2,1,p); plot(1:3); title('T'); end
%% sp_title_colorbar
subplot(2,1,1); plot(1:3); subplot(2,1,2); plot(1:3); title('T'); colorbar
%% sp_title_empty
subplot(2,1,1); plot(1:3); subplot(2,1,2); plot(1:3); title('T'); title('')
%% sp_axes_fontsize
subplot(2,1,1); plot(1:3); subplot(2,1,2); plot(1:3); title('T'); set(gca,'FontSize',16)
%% sp_diferansiyel
t=linspace(0,6,50)'; x=[exp(-t/3).*cos(2*pi*t), -exp(-t/3).*sin(2*pi*t)];
subplot(2,1,1); plot(t,x(:,1),'LineWidth',1.5); grid on;
xlabel('Zaman (s)'); ylabel('Konum'); title('İkinci dereceden sistem');
subplot(2,1,2); plot(x(:,1),x(:,2),'LineWidth',1.5); grid on;
xlabel('Konum'); ylabel('Hız'); title('Faz portresi');
%% sp_doyum
t=0:3; subplot(2,1,1); plot(t,t,'k--',t,[t' 2*t']); grid on; ylabel('Çıkış'); title('Hedef düşürülünce integral yığılmasının etkisi');
subplot(2,1,2); plot(t,[t' 2*t']); grid on; xlabel('Zaman (s)'); ylabel('Aktüatör'); legend('Korumasız PI','Clamping PI');
%% sp_hosgeldin
set(gcf,'Position',[0 0 1000 400]); subplot(1,2,1); plot(1:3); xlabel('Zaman (s)'); ylabel('Genlik'); title('İki frekans, tek sinyal'); grid on;
subplot(1,2,2); stem(1:3,'Color',[0.9 0.4 0.15]); xlim([0 40]); xlabel('Frekans (Hz)'); ylabel('Genlik'); title('FFT · 5 Hz + 20 Hz'); grid on;
%% cb_equal_tight_subplot
subplot(1,2,1); imagesc(magic(4)); axis equal tight; colorbar; title('A matrisi')
%% cb_equal_tight
imagesc(magic(4)); axis equal tight; colorbar
%% cb_tall_equal_tight
imagesc(reshape(1:30,10,3)); axis equal tight; colorbar
%% cb_wide_south
imagesc(reshape(1:30,3,10)); axis equal tight; colorbar('southoutside')
%% cb_tall_west
imagesc(reshape(1:30,10,3)); axis equal tight; colorbar('westoutside')
%% cb_after_equal
imagesc(reshape(1:30,10,3)); colorbar; axis equal tight
%% cb_axis_image
imagesc(reshape(1:30,10,3)); axis image; colorbar
%% cb_daspect
plot(1:3); daspect([1 1 1]); colorbar
%% cb_pbaspect
plot(1:3); pbaspect([1 2 1]); colorbar
%% cmap_turbo
surf(peaks(8)); colormap turbo; colorbar
%% cmap_hsv_pink
imagesc(magic(5)); colormap(hsv(16)); colormap pink
