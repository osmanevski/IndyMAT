% Oracle cases for indy-graphics. Each case starts with "%% name"; the body is
% MATLAB code run in a fresh invisible figure. motor/scripts/grafik_oracle.sh
% runs every case in octave-cli (fltk) through IndyMAT's own extractor
% octave/__mf_figure_data__.m and writes <name>.json next to this file.
% tests/oracle.rs rebuilds the same figure through the Rust API and compares.

%% line_basic
plot([1 2 4],[3 5 4],'-o')
%% line_multi
x=0:0.5:3; plot(x,x.^2,x,2*x,'r--')
%% line_matrix
plot([1 2;3 4;5 7])
%% line_nan
plot([1 2 NaN 4 5],[1 3 2 5 4])
%% line_hold
plot(1:5,[2 4 3 5 1]); hold on; plot(1:5,[1 2 3 4 5],'k:'); hold off
%% line_labels
plot(0:10,(0:10).^2); xlabel('x'); ylabel('y'); title('Kare'); grid on
%% line_legend
plot(1:3,[1 2 3],1:3,[3 2 1]); legend('a','b')
%% line_legend_location
plot(1:3); legend('d','Location','southwest')
%% line_limits
plot(1:10); xlim([0 20]); ylim([-5 15])
%% line_spec_markers
plot(1:4,'g^:','LineWidth',2,'MarkerSize',8)
%% line_sin
x=-3:0.1:3; plot(x,sin(x))
%% line_large
x=linspace(0,1000,50); plot(x,sqrt(x))
%% line_small
plot([0.001 0.002 0.004],[1e-4 3e-4 2e-4])
%% line_set
h=plot(1:3); set(h,'Color',[0.5 0.5 0.5],'LineStyle','--')
%% line_object
line([0 1],[0 1])
%% axis_tight
x=linspace(0,7,30); plot(x,sin(x)); axis tight
%% axis_equal
plot([0 3],[0 1]); axis equal
%% axis_off
plot(1:3); axis off
%% axis_manual
plot(1:3); axis([0 4 0 10])
%% plot3_basic
t=0:0.5:5; plot3(cos(t),sin(t),t)
%% plot3_labels
plot3([0 1 2],[0 1 0],[0 0 1]); xlabel('X'); ylabel('Y'); zlabel('Z'); grid on
%% scatter_basic
scatter([1 2 3 4],[4 1 3 2])
%% scatter_sizes
scatter([1 2 3],[1 4 9],[10 40 90],'r','filled')
%% scatter3_basic
scatter3([1 2 3],[3 1 2],[0 1 4])
%% bar_basic
bar([3 5 2 4])
%% bar_grouped
bar([1 2;3 4;5 6])
%% bar_stacked
bar([1 2;3 4],'stacked')
%% barh_basic
barh([2 4 1])
%% stairs_basic
stairs([1 3 2 4])
%% stem_basic
stem([1 3 2 4])
%% area_basic
area([1 3 2 4])
%% fill_basic
fill([0 1 0.5],[0 0 1],'r')
%% patch_basic
patch([0 2 2 0],[0 0 1 1],[0 0.5 1])
%% surf_grid
[X,Y]=meshgrid(1:4,1:3); surf(X,Y,X+Y)
%% surf_matrix
surf([1 2 3;2 4 6;3 6 9])
%% mesh_basic
mesh([1 2 3;4 5 6])
%% surf_colorbar
surf([1 2 3;4 5 6]); colorbar
%% surf_colormap
surf([1 2;3 4]); colormap(jet(8)); caxis([0 5])
%% surf_view
surf([1 2;3 4]); view(45,60)
%% contour_basic
contour([1 2 3;2 3 4;3 4 5])
%% contourf_basic
contourf([1 2 3;2 3 4;3 4 5])
%% imagesc_basic
imagesc([1 2;3 4])
%% imagesc_colorbar
imagesc([8 1 6;3 5 7;4 9 2]); colorbar
%% imagesc_gray
imagesc([0 1;2 3]); colormap(gray)
%% imagesc_caxis
imagesc([1 2;3 4]); caxis([0 10])
%% text_basic
plot(1:3); text(2,2,'merhaba')
%% subplot_2x1
subplot(2,1,1); plot(1:3); subplot(2,1,2); plot(3:-1:1)
%% subplot_2x2
for k=1:4, subplot(2,2,k); plot(1:k+1); end
%% clf_replot
plot(1:3); clf; plot(3:-1:1)
%% empty_figure
% no axes at all
%% contour_ring
Z=[0 1 2 1 0;1 3 5 3 1;2 5 9 5 2;1 3 5 3 1;0 1 2 1 0]; contour(Z)
%% contourf_ring
Z=[0 1 2 1 0;1 3 5 3 1;2 5 9 5 2;1 3 5 3 1;0 1 2 1 0]; contourf(Z)
%% contour_saddle_n5
contour([1 0 1;0 1 0;1 0 1],5)
%% contour_xy
contour([0 1 3],[0 2],[1 2 3;3 2 1])
%% contour_levels
Z=[0 1 2 1 0;1 3 5 3 1;2 5 9 5 2;1 3 5 3 1;0 1 2 1 0]; contour(Z,[2 4 6])
%% contourf_levels
Z=[0 1 2 1 0;1 3 5 3 1;2 5 9 5 2;1 3 5 3 1;0 1 2 1 0]; contourf(Z,[2 4 6])
%% contourf_hole
contourf([0 0 0 0 0;0 5 5 5 0;0 5 0 5 0;0 5 5 5 0;0 0 0 0 0],[1 3])
%% contour_hole
contour([0 0 0 0 0;0 5 5 5 0;0 5 0 5 0;0 5 5 5 0;0 0 0 0 0],[1 3])
